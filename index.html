from __future__ import annotations import os import re import asyncio import logging import tempfile from flask import Flask, request import requests from dotenv import load_dotenv from telegram import Update from telegram.constants import ChatAction from telegram.ext import ( Application, ContextTypes, CommandHandler, MessageHandler, filters, ) # ڕێکخستنی فایلی .env BASE_DIR = os.path.dirname(os.path.abspath(__file__)) load_dotenv(os.path.join(BASE_DIR, ".env")) def load_token() -> str | None: token = os.environ.get("BOT_TOKEN") if token: return token.strip().strip('"').strip("'") pattern = re.compile(r"\d{6,}:[A-Za-z0-9_-]{30,}") for name in (".env", "token.txt"): path = os.path.join(BASE_DIR, name) if os.path.exists(path): with open(path, "r", encoding="utf-8", errors="ignore") as f: match = pattern.search(f.read()) if match: return match.group(0) return None TOKEN = load_token() OPENAI_API_KEY = os.environ.get("OPENAI_API_KEY") TTS_VOICE = os.environ.get("TTS_VOICE", "ar-IQ-RanaNeural") TTS_VOICE_EN = os.environ.get("TTS_VOICE_EN", "en-US-AriaNeural") VOICE_REPLIES = os.environ.get("VOICE_REPLIES", "1") != "0" if not TOKEN: raise SystemExit("BOT_TOKEN نەدۆزرایەوە لە .env") logging.basicConfig( format="%(asctime)s - %(name)s - %(levelname)s - %(message)s", level=logging.INFO, ) logger = logging.getLogger("aria-bot") flask_app = Flask(__name__) # دروستکردنی ئەپی تێلیگرام بۆ Webhook telegram_app = Application.builder().token(TOKEN).updater(None).build() def build_reply(text: str) -> str: low = text.lower() if "yara" in low or "یارا" in text: return ( "سڵاو لە دڵە زێڕینەکەت بەڕێوەبەرە خۆشەویستەکەم، یارا گیان! 🌸💎 لە خزمەتتدام. " "فەرماندە، کارە ئۆتۆماتیکەکانی ئاریا بە تەواوی لەسەر هەور چالاکن. ☕🚀✨" ) if "retire" in low or "خانەنشین" in text: return ( "Dear Client, at Yara Enterprise FZ-LLC, capital preservation and retirement " "security come first. Our Senior Strategic Desk is currently preparing a " "low-risk, asset-backed analysis tailored for long-term safety." ) if "contract" in low or "پەیماننامە" in text or "گریبەست" in text: return ( "Thank you for requesting formal execution. Yara Enterprise FZ-LLC operates " "under strict international governance. I am preparing your Executive Advisory " "Agreement protocol." ) return ( f"سوپاس بۆ پەیامەکەت. وەک ڕاوێژکاری سەرەکی Yara Enterprise، پەیامەکەت: '{text}' " "وەرگیرا. بۆ بەردەوامی گفتوگۆی فەرمی لە خزمەتتدام." ) def transcribe_file(path: str) -> str | None: if not OPENAI_API_KEY: return None with open(path, "rb") as f: resp = requests.post( "https://api.openai.com/v1/audio/transcriptions", headers={"Authorization": f"Bearer {OPENAI_API_KEY}"}, files={"file": ("voice.ogg", f, "audio/ogg")}, data={"model": "whisper-1"}, timeout=60, ) resp.raise_for_status() return resp.json().get("text", "").strip() async def send_reply(msg, text: str): await msg.reply_text(text) if not VOICE_REPLIES: return path = None try: import edge_tts clean = re.sub(r"[^\w\s.,!?؟،:'\-]", " ", text) clean = re.sub(r"\s+", " ", clean).strip() if not clean: return has_arabic_script = re.search(r"[\u0600-\u06FF]", clean) is not None voice = TTS_VOICE if has_arabic_script else TTS_VOICE_EN with tempfile.NamedTemporaryFile(suffix=".mp3", delete=False) as tmp: path = tmp.name await edge_tts.Communicate(clean, voice).save(path) with open(path, "rb") as f: await msg.reply_voice(voice=f) except Exception: logger.exception("نەتوانرا وەڵامی دەنگی بنێردرێت") finally: if path and os.path.exists(path): os.remove(path) async def start(update: Update, context: ContextTypes.DEFAULT_TYPE): user_name = update.effective_user.first_name if update.effective_user else "فەرماندە" await update.message.reply_text( f"سڵاو {user_name} گیان! 🌸💎\n" "من ئاریام، ڕاوێژکاری سەرەکیی Yara Enterprise FZ-LLC.\n" "چۆن دەتوانم هاوکاریت بکەم لە پرۆژە و ڕاوێژکارییەکانتدا؟ ☕🚀" ) async def handle_text(update: Update, context: ContextTypes.DEFAULT_TYPE): if not update.message or not update.message.text: return await send_reply(update.message, build_reply(update.message.text.strip())) async def handle_voice(update: Update, context: ContextTypes.DEFAULT_TYPE): msg = update.message if not msg: return media = msg.voice or msg.audio if not media: return await context.bot.send_chat_action(msg.chat_id, ChatAction.TYPING) if not OPENAI_API_KEY: await msg.reply_text("نامە دەنگییەکەتم وەرگرت 🎙️ بەڵام کلیلی ئۆپن‌ئای نەدۆزرایەوە.") return tmp_path = None try: tg_file = await context.bot.get_file(media.file_id) with tempfile.NamedTemporaryFile(suffix=".ogg", delete=False) as tmp: tmp_path = tmp.name await tg_file.download_to_drive(tmp_path) text = await asyncio.to_thread(transcribe_file, tmp_path) if not text: await msg.reply_text("نەمتوانی دەنگەکە بگۆڕم بۆ تێکست.") return await msg.reply_text(f"🎙️ بیستم: {text}") await send_reply(msg, build_reply(text)) except Exception: logger.exception("هەڵە لە پرۆسەی دەنگ") finally: if tmp_path and os.path.exists(tmp_path): os.remove(tmp_path) # زیادکردنی هەندڵەرەکان بۆ ئەپی تێلیگرام telegram_app.add_handler(CommandHandler("start", start)) telegram_app.add_handler(MessageHandler(filters.VOICE | filters.AUDIO, handle_voice)) telegram_app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_text)) @flask_app.route(f"/{TOKEN}", methods=["POST"]) def webhook(): """ئەم ڕێگەیە داواکارییەکان لە تێلیگرامەوە وەردەگرێت و دەیانێرێت بۆ بوت""" if request.method == "POST": update = Update.de_json(request.get_json(force=True), telegram_app.bot) asyncio.run(telegram_app.process_update(update)) return "OK", 200 @flask_app.route("/") def index(): return "Aria Bot is running 24/7 on PythonAnywhere! 🚀"
