import os import asyncio import logging import tempfile import requests from dotenv import load_dotenv from telegram import Update from telegram.constants import ChatAction from telegram.ext import ( ApplicationBuilder, ContextTypes, CommandHandler, MessageHandler, filters, ) # فایلی .env لە هەمان فۆڵدەری ئەم فایلە بخوێنەوە (ڕێگای تەواو، تا لە PythonAnywhere کار بکات) BASE_DIR = os.path.dirname(os.path.abspath(__file__)) load_dotenv(os.path.join(BASE_DIR, ".env")) TOKEN = os.environ.get("BOT_TOKEN") OPENAI_API_KEY = os.environ.get("OPENAI_API_KEY") # ئارەزوومەندانەیە، بۆ گۆڕینی دەنگ بۆ تێکست if not TOKEN: raise SystemExit("BOT_TOKEN نەدۆزرایەوە. لە فایلی .env بینووسە: BOT_TOKEN=...") logging.basicConfig( format="%(asctime)s - %(name)s - %(levelname)s - %(message)s", level=logging.INFO, ) logger = logging.getLogger("aria-bot") def build_reply(text: str) -> str: """لۆجیکی وەڵامدانەوە، هاوبەشە لە نێوان تێکست و دەنگ.""" low = text.lower() if "yara" in low or "یارا" in text: return ( "سڵاو لە دڵە زێڕینەکەت بەڕێوەبەرە خۆشەویستەکەم، یارا گیان! 🌸💎 لە خزمەتتدام. " "فەرماندە، کارە ئۆتۆماتیکەکانی ئاریا بە تەواوی چالاکن. ☕🚀✨" ) if "retire" in low or "خانەنشین" in text: return ( "Dear Client, at Yara Enterprise FZ-LLC, capital preservation and retirement " "security come first. Our Senior Strategic Desk is currently preparing a " "low-risk, asset-backed analysis tailored for long-term safety." ) if "contract" in low or "پەیماننامە" in text or "گریبەست" in text: return ( "Thank you for requesting formal execution. Yara Enterprise FZ-LLC operates " "under strict international governance. I am preparing your Executive Advisory " "Agreement protocol." ) return ( f"سوپاس بۆ پەیامەکەت. وەک ڕاوێژکاری سەرەکی Yara Enterprise، پەیامەکەت: '{text}' " "وەرگیرا. بۆ بەردەوامی گفتوگۆی فەرمی لە خزمەتتدام." ) def transcribe_file(path: str) -> str | None: """دەنگ دەگۆڕێت بۆ تێکست بە Whisper. ئەگەر کلیل نەبێت None دەگەڕێنێتەوە.""" if not OPENAI_API_KEY: return None with open(path, "rb") as f: resp = requests.post( "https://api.openai.com/v1/audio/transcriptions", headers={"Authorization": f"Bearer {OPENAI_API_KEY}"}, files={"file": ("voice.ogg", f, "audio/ogg")}, data={"model": "whisper-1"}, timeout=60, ) resp.raise_for_status() return resp.json().get("text", "").strip() async def start(update: Update, context: ContextTypes.DEFAULT_TYPE): user_name = update.effective_user.first_name if update.effective_user else "فەرماندە" await update.message.reply_text( f"سڵاو {user_name} گیان! 🌸💎\n" "من ئاریام، ڕاوێژکاری سەرەکیی Yara Enterprise FZ-LLC.\n" "چۆن دەتوانم هاوکاریت بکەم لە پرۆژە و ڕاوێژکارییەکانتدا؟ ☕🚀" ) async def handle_text(update: Update, context: ContextTypes.DEFAULT_TYPE): if not update.message or not update.message.text: return await update.message.reply_text(build_reply(update.message.text.strip())) async def handle_voice(update: Update, context: ContextTypes.DEFAULT_TYPE): msg = update.message if not msg: return media = msg.voice or msg.audio if not media: return await context.bot.send_chat_action(msg.chat_id, ChatAction.TYPING) if not OPENAI_API_KEY: await msg.reply_text( "نامە دەنگییەکەتم وەرگرت 🎙️ بەڵام هێشتا ناتوانم دەنگ بگۆڕم بۆ تێکست. " "تکایە پەیامەکەت بە نووسین بنێرە. 🌸" ) return tmp_path = None try: tg_file = await context.bot.get_file(media.file_id) with tempfile.NamedTemporaryFile(suffix=".ogg", delete=False) as tmp: tmp_path = tmp.name await tg_file.download_to_drive(tmp_path) # کارە درێژخایەنەکە لە thread جیا بکە تا بوتەکە نەوەستێت text = await asyncio.to_thread(transcribe_file, tmp_path) if not text: await msg.reply_text("نەمتوانی دەنگەکە بگۆڕم بۆ تێکست. تکایە دووبارە بنێرە یان بنووسە. 🌸") return await msg.reply_text(f"🎙️ بیستم: {text}") await msg.reply_text(build_reply(text)) except Exception: logger.exception("هەڵە لە پرۆسەی نامە دەنگییەکە") await msg.reply_text("ببورە، کێشەیەک ڕوویدا لە کاتی پرۆسەی دەنگەکە. تکایە دووبارە هەوڵ بدەرەوە. 🌸") finally: if tmp_path and os.path.exists(tmp_path): os.remove(tmp_path) async def on_error(update: object, context: ContextTypes.DEFAULT_TYPE): logger.error("هەڵەی بێ-گرتن:", exc_info=context.error) if __name__ == "__main__": print("Starting Aria Bot Engine...") app = ApplicationBuilder().token(TOKEN).build() app.add_handler(CommandHandler("start", start)) app.add_handler(MessageHandler(filters.VOICE | filters.AUDIO, handle_voice)) app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_text)) app.add_error_handler(on_error) print("Bot is running... (Aria Online 🚀)") app.run_polling(drop_pending_updates=True)
